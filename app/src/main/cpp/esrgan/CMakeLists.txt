cmake_minimum_required(VERSION 3.22)
if(NOT CMAKE_PROJECT_NAME)
  project(img23d_esrgan LANGUAGES C CXX)
endif()

# ---- ncnn (Tencent/ncnn, BSD-3): pinned prebuilt release, downloaded once and checked against its SHA-256 ----
set(NCNN_VERSION 20261020)
if(ANDROID)
  set(NCNN_ZIP "ncnn-${NCNN_VERSION}-android-vulkan.zip")
  set(NCNN_SHA256 "6e0c5f9b397faf32b923de2ef6210af9bf375851869dceff06a8005331981086")
  set(NCNN_PREFIX_REL "ncnn-${NCNN_VERSION}-android-vulkan/${ANDROID_ABI}")
else()
  set(NCNN_ZIP "ncnn-${NCNN_VERSION}-ubuntu-2204.zip")
  set(NCNN_SHA256 "0e364326f6b6e225f1d1886e2127898c9d3383811a894e2680d156f589fe2641")
  set(NCNN_PREFIX_REL "ncnn-${NCNN_VERSION}-ubuntu-2204")
endif()
if(DEFINED ENV{NCNN_CACHE_DIR})
  set(NCNN_CACHE "$ENV{NCNN_CACHE_DIR}")
else()
  set(NCNN_CACHE "${CMAKE_CURRENT_LIST_DIR}/../../../../build/ncnn-cache")
endif()
get_filename_component(NCNN_CACHE "${NCNN_CACHE}" ABSOLUTE)
set(NCNN_ROOT "${NCNN_CACHE}/${NCNN_VERSION}-${ANDROID_ABI}")
if(NOT EXISTS "${NCNN_ROOT}/${NCNN_PREFIX_REL}/lib/cmake/ncnn/ncnnConfig.cmake")
  set(_zip "${NCNN_CACHE}/${NCNN_ZIP}")
  file(MAKE_DIRECTORY "${NCNN_CACHE}")
  if(NOT EXISTS "${_zip}")
    string(RANDOM LENGTH 8 _rnd)
    message(STATUS "Downloading ${NCNN_ZIP}")
    file(DOWNLOAD "https://github.com/Tencent/ncnn/releases/download/${NCNN_VERSION}/${NCNN_ZIP}" "${_zip}.${_rnd}"
         EXPECTED_HASH SHA256=${NCNN_SHA256} SHOW_PROGRESS TLS_VERIFY ON)
    file(RENAME "${_zip}.${_rnd}" "${_zip}")
  endif()
  file(ARCHIVE_EXTRACT INPUT "${_zip}" DESTINATION "${NCNN_ROOT}")
endif()
set(ncnn_DIR "${NCNN_ROOT}/${NCNN_PREFIX_REL}/lib/cmake/ncnn")
find_package(ncnn REQUIRED)

# ---- Real-ESRGAN core (plain C++; also built by the host test) ----
add_library(esrgan_core STATIC esrgan_core.cpp)
target_include_directories(esrgan_core PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_compile_features(esrgan_core PUBLIC cxx_std_17)
target_link_libraries(esrgan_core PUBLIC ncnn)

if(ANDROID)
  add_library(img23d_esrgan SHARED jni.cpp ndk_compat.cpp)
  target_link_libraries(img23d_esrgan PRIVATE esrgan_core log)
  # drop unused ncnn code (layers/shaders of other models stay linked by ncnn's registry, but the rest goes)
  target_compile_options(img23d_esrgan PRIVATE -ffunction-sections -fdata-sections)
  target_link_options(img23d_esrgan PRIVATE -Wl,-z,max-page-size=16384 -Wl,--gc-sections)
endif()
